        GeoLocObj["[Geolocation Request] Service Area Validation & Zone Mapping"]
        %% Core Entity: Cart Item
        CartItem["[Cart Item] Inventory Check & Price Calculation"]
        %% Output
        DeliveryPin["DeliveryPin[Confirmed Delivery Pin & Fee]"]
        CartState["CartState[Updated Cart State w/ Stock]"]
        
        %% Internal Processing
        GeoLocObj --"Validates Zone"--> DeliveryPin
        DeliveryPin --"Fee Applied"--> CartItem
        CartItem --"Real-time Check"--> CartState
        CartState --"Validates Selection"--> ExitNode2["ExitNode2: Staged Order (Cart) Ready"]
    end
    
    %% =========================================================================
    %% PHASE 3: SLOT ALLOCATION & ORDER COMMIT
    %% =========================================================================
    subgraph SLOT_COMMITMENT_GATEWAY["Phase 3: Slot Allocation & Commitment"]
        direction TB
        %% Entrance
        ExitNode2 --"Triggers Request"--> SlotReq["SlotReq{User Requests Delivery Slot}"]
        %% Core Entity: Delivery Slot
        SlotObj["[Delivery Slot] Allocation & Time-Window Enforcement"]
        %% Core Entity: Order Header
        OrderHeader["[Order Header] Creation & Auth Init"]
        %% Core Entity: Order Line Item
        OrderLineItem["[Order Line Item] Stock Reservation & Weight Est"]
        %% Core Entity: Payment Transaction
        PaymentTx["[Payment Transaction] Fund Transfer & Receipt Gen"]
        %% Output
        OrderID["OrderID[Unique Order ID & Receipt]"]
        
        %% Logic: Authorization Check
        AuthCheck{"Is Payment Authorized?"}
        AuthSuccess["AuthSuccess[Authorized Status Achieved]"]
        AuthFail["AuthFail[Authorization Rejected]"]
        
        %% Logic: Document Creation Check
        DocCheck{"Is Settlement Complete?"}
        DocCreate["DocCreate[Post-Settlement Document Created]"]
        
        %% Internal Processing
        SlotObj --"Allocates Time"--> OrderHeader
        OrderHeader --"Captures Header"--> OrderLineItem
        OrderLineItem --"Reserves Stock"--> PaymentTx
        PaymentTx --"Processes Funds"--> AuthCheck
        
        AuthCheck --"Yes"--> AuthSuccess
        AuthCheck --"No"--> AuthFail
        
        AuthSuccess --"Enables Docs"--> DocCheck
        DocCheck --"Yes"--> DocCreate
        DocCheck --"No"--> AuthFail
        
        DocCreate --"Generates Order"--> OrderID
        AuthFail --"Blocks Commit"--> OrderID
        
        OrderID --"Confirms Receipt"--> ExitNode3["ExitNode3: Finalized Order Ready"]
    end
    
    %% =========================================================================
    %% PHASE 4: LOGISTICS FULFILLMENT & HANDOVER
    %% =========================================================================
    subgraph LOGISTICS_FULFILLMENT_GATEWAY["Phase 4: Logistics & Handover"]
        direction TB
        %% Entrance
        ExitNode3 --"Dispatches Task"--> CourierAssign["CourierAssign{Dispatcher Assigns Courier}"]
        %% Core Entity: Assigned Courier
        CourierObj["[Assigned Courier] Route Optimization & Task Active"]
        %% Core Entity: Pickup Manifest
        PickupManifest["[Pickup Manifest] Store Task Gen & Verification"]
        %% Core Entity: Merchant Inventory Stock
        InvStock["[Merchant Inventory Stock] Batch Deduction & Shelf Alloc"]
        %% Core Entity: Handover Receipt
        HandoverRec["[Handover Receipt] Completion Validation & Quality Check"]
        %% Output
        CourierID["CourierID[Active Courier ID & Location]"]
        FinalStatus["FinalStatus[Delivered & Payment Captured]"]
        
        %% Logic: Exception Check (Mid-Flight)
        ExceptionCheck{"Did Exception Occur?"}
        ExceptionRollback["ExceptionRollback[Reversing Ledger Action]"]
        ExceptionClosure["ExceptionClosure[System Disruption Closed]"]
        
        %% Internal Processing
        CourierAssign --"Optimizes Route"--> CourierObj
        CourierObj --"Activates Task"--> PickupManifest
        PickupManifest --"Generates Task"--> InvStock
        InvStock --"Deducts Stock"--> HandoverRec
        HandoverRec --"Validates Quality"--> ExceptionCheck
        
        ExceptionCheck --"Yes"--> ExceptionRollback
        ExceptionCheck --"No"--> CourierID
        
        ExceptionRollback --"Rolls Back Funds"--> ExceptionClosure
        CourierID --"Tracks Transit"--> FinalStatus
        FinalStatus --"Confirms Delivery"--> ExitNode4["ExitNode4: Transaction Completed"]
    end
    
    %% =========================================================================
    %% PHASE 5: POST-ORDER LIFECYCLE & CLOSURE
    %% =========================================================================
    subgraph POST_ORDER_CLOSURE_GATEWAY["Phase 5: Post-Order Lifecycle"]
        direction TB
        %% Entrance
        ExitNode4 --"Enables Options"--> PostOrderOpts["PostOrderOpts{User: Refund, Cancel, or Loyalty}"]
        %% Core Entity: Loyalty Points Ledger
        LoyaltyPoints["[Loyalty Points Ledger] Point Accrual & Redemption"]
        %% Core Entity: Refund Claim
        RefundClaim["[Refund Claim] Claim Validation & Fund Reversal"]
        %% Core Entity: Cancellation Request
        CancelReq["[Cancellation Request] Inventory Release Logic"]
        %% Output
        AdjustedBalance["AdjustedBalance[Updated Account Balance]"]
        ReleasedStock["ReleasedStock[Released Stock Availability]"]
        
        %% Internal Processing
        PostOrderOpts --"Selects Loyalty"--> LoyaltyPoints
        PostOrderOpts --"Selects Refund"--> RefundClaim
        PostOrderOpts --"Selects Cancel"--> CancelReq
        
        LoyaltyPoints --"Calculates Points"--> AdjustedBalance
        RefundClaim --"Reverses Funds"--> AdjustedBalance
        CancelReq --"Releases Stock"--> ReleasedStock
        
        ReleasedStock --"Restores Inventory"--> ExitNode5["ExitNode5: System Closed"]
        AdjustedBalance --"Updates Ledger"--> ExitNode5
    end
    
    %% =========================================================================
    %% CROSS-REFERENCE & EXCEPTION FLOW (MANDATORY)
    %% =========================================================================
    
    %% Exception Path: Post-Settlement Failure (Rollback)
    subgraph EXCEPTION_HANDLER["Exception Handler: Mid-Flight Rollback"]
        direction TB
        ExceptionRollback --"Initiates Compensating Transaction"--> CompensatingTx["CompensatingTx[Compensating Transaction Executed]"]
        CompensatingTx --"Reverses Ledger"--> ExceptionClosure
    end
    
    %% Linking Exceptions to Main Flow
    ExitNode4 --"|4.2-A. Post-Settlement Failure|"-."->"ExceptionRollback"
    ExitNode4 --"|4.2-B. Success|"--> PostOrderOpts
    
    %% Linking Cancellation (Inventory Release)
    PostOrderOpts --"|5.1. Cancel Order|"--> CancelReq
    CancelReq --"Triggers Release"--> MerchantInvRelease["MerchantInvRelease[Inventory Release Logic]"]
    MerchantInvRelease --"Updates Stock"--> ReleasedStock
    
    %% Linking Refunds (Finance)
    PostOrderOpts --"|5.2. Refund Claim|"--> FinanceRefundCheck{"Is Claim Valid?"}
    FinanceRefundCheck --"Yes"--> RefundClaim
    FinanceRefundCheck --"No"--> RefundReject["RefundReject[Claim Rejected]"]
    
    %% Linking Loyalty (User)
    PostOrderOpts --"|5.3. Redeem Points|"--> LoyaltyPoints
    
    %% Linking Cancellation (Merchant)
    CancelReq --"Triggers Release"--> MerchantInvRelease
    MerchantInvRelease --"Updates Stock"--> ReleasedStock
    
    %% =========================================================================
    %% FINAL CLOSURE NODES
    %% =========================================================================
    ExitNode5 --> EndState["EndState[Transaction Lifecycle Closed]"]
    ExceptionClosure --> EndState
    RefundReject --> EndState
